--1/c  MasikPiros_Auto.sql
CREATE TABLE MasikPiros_Auto (
    rsz VARCHAR2(20),
    tipus VARCHAR2(20),
    szin VARCHAR2(20),
    kor NUMBER(3),
    ar NUMBER(10)
);




--d.) Kurzor használatával írjon egy PL/SQL blokkot a piros autók lekérdezésére és külön táblában (MasikPiros_Auto) való eltárolására!
--Mentés: PirosAuto.sql
--A kurzor neve: piros; változó név: x
--A PirosAuto.sql sckript lefuttatása után, a MasikPiros_Auto tábla frissítése, kilistázza a piros autókat




SET SERVEROUTPUT ON;

DECLARE
    CURSOR piros IS SELECT rsz, tipus, szin, kor, ar FROM Piros_Auto WHERE szin = 'piros';
        

    x piros%ROWTYPE;     -- Változó deklarálása a kurzor egy sorának tárolására

BEGIN
    OPEN piros;
    
    LOOP
       FETCH piros INTO x;
       EXIT WHEN piros%NOTFOUND;
       INSERT INTO MasikPiros_Auto (rsz, tipus, szin, kor, ar) VALUES (x.rsz, x.tipus, x.szin, x.kor, x.ar);
    END LOOP;
    CLOSE piros;
END;
SELECT * FROM MasikPiros_Auto;





--1/e PirosAutoFor.sql
SET SERVEROUTPUT ON;

DECLARE
    -- Kurzor deklarálása a piros autók lekérdezésére
    CURSOR piros_kurzor IS SELECT rsz, tipus, szin, kor, ar FROM Piros_Auto WHERE szin = 'piros';
BEGIN
    FOR auto_rekord IN piros_kurzor LOOP
        INSERT INTO MasikPiros_Auto (rsz, tipus, szin, kor, ar) 
        VALUES (auto_rekord.rsz, auto_rekord.tipus, auto_rekord.szin, auto_rekord.kor, auto_rekord.ar);
    END LOOP;
END;
SELECT * FROM MasikPiros_Auto;
